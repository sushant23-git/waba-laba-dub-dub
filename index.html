Practical 1: Transfer Functions



clc;
clear;
close all;

s=tf('s');

disp('Individual Transfer Functions');

G1=1/(s+1)
G2=2/(s+2)

disp('Series Connection')

G_series=series(G1,G2)

disp('Parallel Connection')

G_parallel=parallel(G1,G2)

disp('Feedback Connection')

G_feedback=feedback(G1,G2)

Practical 2: Step Response



clc;
clear;
close all;

wn=5;
zeta=0.5;

sys=tf(wn^2,[1 2*zeta*wn wn^2])

[y,t]=step(sys);

figure
plot(t,y,'LineWidth',3)

grid on

title('Unit Step Response')
xlabel('Time')
ylabel('Amplitude')

info=stepinfo(sys)

disp('Rise Time')
disp(info.RiseTime)

disp('Peak Time')
disp(info.PeakTime)

disp('Settling Time')
disp(info.SettlingTime)

disp('Overshoot')
disp(info.Overshoot)

Practical 3: Steady State Error



clc
clear
close all

s=tf('s');

G=10/(s*(s+2))

kp=dcgain(G)

kv=dcgain(s*G)

ka=dcgain((s^2)*G)

ess_step=1/(1+kp)

ess_ramp=1/kv

ess_parabolic=1/ka

disp(kp)
disp(kv)
disp(ka)

disp(ess_step)
disp(ess_ramp)
disp(ess_parabolic)

Practical 4: Parameter Variation



clc
clear
close all

w=5
z=0.5

figure
hold on

K=[1 5 10]

for i=1:length(K)

sys=tf(K(i)*w^2,[1 2*z*w K(i)*w^2])

step(sys)

end

legend('1','5','10')

Practical 6: Lead Compensation



clc
clear
close all

G=tf(10,[1 1 0])

Gc=0.9*tf([1 1],[1 3])

T1=feedback(G,1)

T2=feedback(series(Gc,G),1)

figure

step(T1,'r',T2,'b')

legend('Without Compensation','Lead Compensation')

grid on

Practical 8: State Space



clc
clear
close all

den=[1 5 6]

[A,B,C,D]=tf2ss(num,den)

sys=ss(A,B,C,D)

step(G,'r',G2,'b')
